@uniform_row_sample_norm_tail : ∀ {N m : ℕ},
  0 < N →
    0 < m →
      ∀ (hm : m ≤ N) (F : Matrix (Fin N) (Fin N) ℝ) (d : ℝ),
        0 < d →
          (∀ (i : Fin N), rowEnergy F i ≤ d) →
            ∀ (p : ↑unitInterval),
              ↑p * ↑N = ↑m →
                ∀ (t : ℝ),
                  0 < t →
                    (uniformFixedColumnLaw N m hm).real {ω | ‖rowSample F (selectedColumns ω)‖ ≥ √↑p * ‖F‖ + √d * √t} ≤
                      2 * ↑N * (↑N + 1) * Real.exp (-t)
