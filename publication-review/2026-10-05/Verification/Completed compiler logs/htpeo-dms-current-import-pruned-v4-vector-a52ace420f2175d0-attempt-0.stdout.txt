structure MGraph : Type
number of parameters: 0
fields:
  MGraph.n : ℕ
  MGraph.m : ℕ
  MGraph.ends : Fin self.m → Fin self.n × Fin self.n
constructor:
  MGraph.mk (n m : ℕ) (ends : Fin m → Fin n × Fin n) : MGraph
def MGraph.Star : {G : MGraph} → (k : ℕ) → (Fin G.m → Fin k) → Prop :=
fun {G} k c => MGraph.StarOn (fun x => True) k c
def MGraph.StarOn : {G : MGraph} → (Fin G.m → Prop) → (k : ℕ) → (Fin G.m → Fin k) → Prop :=
fun {G} P k c =>
  (∀ (a b : Fin G.m), G.Adj a b → P a → P b → c a ≠ c b) ∧
    ∀ (w : G.Walk4), P w.e1 → P w.e2 → P w.e3 → P w.e4 → ¬MGraph.Bicol c w
def MGraph.Bicol : {G : MGraph} → {k : ℕ} → (Fin G.m → Fin k) → G.Walk4 → Prop :=
fun {G} {k} c w => c w.e1 = c w.e3 ∧ c w.e2 = c w.e4

def MGraph.Subcubic : MGraph → Prop :=
fun G =>
  ∀ (x : Fin G.n) (a b c d : Fin G.m),
    G.Inc a x → G.Inc b x → G.Inc c x → G.Inc d x → a ≠ b → a ≠ c → a ≠ d → b ≠ c → b ≠ d → c ≠ d → False
def MGraph.Loopless : MGraph → Prop :=
fun G => ∀ (f : Fin G.m), (G.ends f).1 ≠ (G.ends f).2
def MGraph.Colourable : {G : MGraph} → (Fin G.m → Prop) → ℕ → Prop :=
fun {G} P k => ∃ c, MGraph.StarOn P k c
def RH2F.DMS : Prop :=
∀ (G : MGraph), G.Subcubic → G.Loopless → ∀ (P : Fin G.m → Prop), MGraph.Colourable P 6
def BlockStar.DMSfor : MGraph → Prop :=
fun G => G.Subcubic → G.Loopless → ∀ (P : Fin G.m → Prop), MGraph.Colourable P 6
def StarFamily : MGraph → ℕ → Prop :=
fun G k =>
  G.Subcubic ∧
    G.Loopless ∧ (∀ (P : Fin G.m → Prop), MGraph.Colourable P k) ∧ ∀ (P : Fin G.m → Prop), MGraph.Colourable P 6
def GPk.StarFam : MGraph → ℕ → Prop :=
fun G k =>
  G.Subcubic ∧
    G.Loopless ∧ (∀ (P : Fin G.m → Prop), MGraph.Colourable P k) ∧ ∀ (P : Fin G.m → Prop), MGraph.Colourable P 6
structure BlockStar.Wir (W : Type) (r q : ℕ) : Type
number of parameters: 3
fields:
  BlockStar.Wir.D : ℕ
  BlockStar.Wir.src : W → ℕ → ℕ
  BlockStar.Wir.jmp : W → ℕ → ℕ
  BlockStar.Wir.dst : W → ℕ → ℕ
  BlockStar.Wir.hsrc : ∀ (w : W) (s : ℕ), self.src w s < r
  BlockStar.Wir.hdst : ∀ (w : W) (s : ℕ), self.dst w s < r
  BlockStar.Wir.hjmp : ∀ (w : W) (s : ℕ), self.jmp w s ≤ self.D
  BlockStar.Wir.jumps : List ℕ
  BlockStar.Wir.hjmps : ∀ (w : W) (s : ℕ), self.jmp w s ∈ self.jumps
constructor:
  BlockStar.Wir.mk {W : Type} {r q : ℕ} (D : ℕ) (src jmp dst : W → ℕ → ℕ) (hsrc : ∀ (w : W) (s : ℕ), src w s < r)
    (hdst : ∀ (w : W) (s : ℕ), dst w s < r) (hjmp : ∀ (w : W) (s : ℕ), jmp w s ≤ D) (jumps : List ℕ)
    (hjmps : ∀ (w : W) (s : ℕ), jmp w s ∈ jumps) : BlockStar.Wir W r q
def BlockStar.BG : {W : Type} → {r q : ℕ} → BlockStar.Wir W r q → ℕ → (ℕ → W) → MGraph :=
fun {W} {r q} Wr n ω =>
  { n := n * r, m := n * q,
    ends := fun e =>
      (⟨↑e / q * r + Wr.src (ω (↑e / q)) (↑e % q), ⋯⟩,
        ⟨(↑e / q + Wr.jmp (ω (↑e / q)) (↑e % q)) % n * r + Wr.dst (ω (↑e / q)) (↑e % q), ⋯⟩) }
def BlockStar.bcol : {C : Type} →
  ℕ → (ct : C → ℕ → ℕ) → {k : ℕ} → (∀ (c : C) (s : ℕ), ct c s < k) → (ℕ → C) → {m : ℕ} → Fin m → Fin k :=
fun {C} q ct {k} hct χ {m} e => ⟨ct (χ (↑e / q)) (↑e % q), ⋯⟩

def Inflation.inflate : (H : MGraph) → (Fin H.m → Bool → Fin 3) → MGraph :=
fun H slot =>
  { n := H.n * 9, m := H.n * 12 + H.m,
    ends := fun f =>
      if h : ↑f < H.n * 12 then
        (Inflation.mkV (↑f / 12) ⋯ (Inflation.pe1 (↑f % 12)), Inflation.mkV (↑f / 12) ⋯ (Inflation.pe2 (↑f % 12)))
      else
        (Inflation.mkV ↑(Inflation.endOf H ⟨↑f - H.n * 12, ⋯⟩ false) ⋯
            (Inflation.portv (slot ⟨↑f - H.n * 12, ⋯⟩ false)),
          Inflation.mkV ↑(Inflation.endOf H ⟨↑f - H.n * 12, ⋯⟩ true) ⋯
            (Inflation.portv (slot ⟨↑f - H.n * 12, ⋯⟩ true))) }
def Inflation.PortsInj : (H : MGraph) → (Fin H.m → Bool → Fin 3) → Prop :=
fun H slot =>
  ∀ (e : Fin H.m) (b : Bool) (e' : Fin H.m) (b' : Bool),
    Inflation.endOf H e b = Inflation.endOf H e' b' → slot e b = slot e' b' → e = e' ∧ b = b'

@BlockStar.star_of_windows : ∀ {W C : Type} {r q : ℕ} (Wr : BlockStar.Wir W r q) (ct : C → ℕ → ℕ) {k : ℕ}
  (hct : ∀ (c : C) (s : ℕ), ct c s < k) (n : ℕ) (ω : ℕ → W) (χ : ℕ → C),
  (∀ j < n,
      ∃ wk ck, (∀ p ≤ 3 * Wr.D, wk p = ω ((j + p) % n) ∧ ck p = χ ((j + p) % n)) ∧ BlockStar.winOK Wr ct wk ck = true) →
    MGraph.Star k (BlockStar.bcol q ct hct χ)
@BlockStar.BG_ends : ∀ {W : Type} {r q : ℕ} (Wr : BlockStar.Wir W r q) (n : ℕ) (ω : ℕ → W) (J s : ℕ),
  J < n →
    s < q →
      ∃ (h : J * q + s < n * q),
        ↑((BlockStar.BG Wr n ω).ends ⟨J * q + s, h⟩).1 = J * r + Wr.src (ω J) s ∧
          ↑((BlockStar.BG Wr n ω).ends ⟨J * q + s, h⟩).2 = (J + Wr.jmp (ω J) s) % n * r + Wr.dst (ω J) s
@BlockStar.BG_loopless : ∀ {W : Type} {r q : ℕ} (Wr : BlockStar.Wir W r q) (n : ℕ) (ω : ℕ → W),
  Wr.D < n → (∀ (w : W), ∀ s < q, Wr.jmp w s = 0 → Wr.src w s ≠ Wr.dst w s) → (BlockStar.BG Wr n ω).Loopless
@BlockStar.BG_subcubic : ∀ {W : Type} {r q : ℕ} (Wr : BlockStar.Wir W r q) (n : ℕ) (ω : ℕ → W),
