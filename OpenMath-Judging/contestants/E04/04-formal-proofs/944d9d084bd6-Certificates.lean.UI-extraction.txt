/-
Certificates for the four schemes. Each `_valid` theorem is closed by `decide`:
the Lean kernel evaluates `brentOK` on the data and checks that the result is
`true`. No `native_decide` (compiler trust) is used anywhere. Each `_identities`
theorem restates the result as the 729 Brent identities through `brentOK_sound`,
and each `_support` theorem counts the nonzero entries.
-/
import MM3.Brent
import MM3.Scheme138
import MM3.Scheme138Gauged
import MM3.Scheme139Exotic
import MM3.Scheme143Core1809


namespace MM3


/-! ## solutions/support_138/solution.json -/


theorem scheme138_valid :
    brentOK Scheme138.su Scheme138.sv Scheme138.sw Scheme138.u Scheme138.v Scheme138.w = true := by
  decide


theorem scheme138_identities :
    ∀ a b c : Nat, a < 9 → b < 9 → c < 9 →
      coeff (terms Scheme138.u Scheme138.v Scheme138.w) a b c =
        Scheme138.su * Scheme138.sv * Scheme138.sw * target a b c :=
  brentOK_sound _ _ _ _ _ _ scheme138_valid


theorem scheme138_support : support Scheme138.u Scheme138.v Scheme138.w = 138 := by
  decide
