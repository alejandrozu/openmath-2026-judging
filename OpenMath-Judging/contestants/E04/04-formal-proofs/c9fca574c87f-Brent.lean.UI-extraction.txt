/-
Brent identities for 3x3 matrix-multiplication schemes.
A scheme of rank R consists of three R x 9 coefficient matrices u, v, w. Term t
computes (sum_e u[t][e] * A_e) * (sum_e v[t][e] * B_e), and output entry f of
C receives sum_t w[t][f] * term_t. Entries of A and B are indexed row-major
(A_e = A[e / 3][e % 3]) and entries of C as 3 * column + row (C_f = C[f % 3][f / 3]).
The scheme computes the product of two 3x3 matrices if and only if, for all
a, b, c in 0..8,
    sum_t u[t][a] * v[t][b] * w[t][c] = T[a][b][c],
where T[3i+j][3j+k][3k+i] = 1 and all other entries of T are 0.
The data files hold integer matrices obtained from the rational scheme by
scaling u, v, w by positive integers su, sv, sw (the least common multiples of
the denominators), so the identity that is checked reads
    sum_t u[t][a] * v[t][b] * w[t][c] = su * sv * sw * T[a][b][c].
Everything below is structurally recursive on lists and natural numbers so
that the kernel can evaluate `brentOK` and `support` by `decide`.
-/


namespace MM3


/-- The rank of the schemes certified here (number of terms). -/
def rank : Nat := 23


/-- The target tensor of 3x3 matrix multiplication: `T a b c = 1` if and only if
`a = 3i + j`, `b = 3j + k`, `c = 3k + i` for some `i j k`, and `0` otherwise. -/
def target (a b c : Nat) : Int :=
  if b / 3 == a % 3 && c / 3 == b % 3 && c % 3 == a / 3 then 1 else 0


/-- Entry `i` of a row, `0` when out of range. -/
def nth : List Int → Nat → Int
  | [], _ => 0
  | x :: _, 0 => x
  | _ :: xs, n + 1 => nth xs n


/-- One triple of rows per term. -/
def terms (u v w : List (List Int)) : List (List Int × List Int × List Int) :=
  u.zip (v.zip w)


/-- The coefficient of `A_a * B_b` in output entry `c`: `sum_t u[t][a] * v[t][b] * w[t][c]`. -/
def coeff (ts : List (List Int × List Int × List Int)) (a b c : Nat) : Int :=
  match ts with
  | [] => 0
  | (ut, vt, wt) :: rest => nth ut a * nth vt b * nth wt c + coeff rest a b c


/-- A matrix has `rank` rows of 9 entries each. -/
def shapeOK (m : List (List Int)) : Bool :=
  m.length == rank && m.all fun row => row.length == 9
